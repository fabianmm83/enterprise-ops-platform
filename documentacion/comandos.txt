Comandos de prueba rápida
cmd
# Login
curl -X POST http://localhost:8080/api/v1/auth/login -H "Content-Type: application/json" -d "{\"email\":\"test@example.com\",\"password\":\"Password123!\"}"

# Guardar token
set TOKEN_ADMIN=<token>

# Listar usuarios (ADMIN)
curl http://localhost:8080/api/v1/users -H "Authorization: Bearer %TOKEN_ADMIN%"

# Listar proyectos
curl http://localhost:8080/api/v1/projects -H "Authorization: Bearer %TOKEN_ADMIN%"

# Mis tareas
curl http://localhost:8080/api/v1/tasks/me -H "Authorization: Bearer %TOKEN_ADMIN%"
